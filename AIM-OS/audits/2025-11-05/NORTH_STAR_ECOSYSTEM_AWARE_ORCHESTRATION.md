```python
def validate_ecosystem_organization(chapter_id: str) -> ValidationResult:
    """
    Validate that the ecosystem is properly organized
    BEFORE writing this chapter.
    """
    
    # Get chapter requirements
    chapter_context = get_chapter_context_mesh(chapter_id)
    
    issues = []
    
    # Check 1: Source docs in correct locations?
    for doc in chapter_context.source_docs:
        expected_path = get_expected_path(doc.system, doc.doc_type)
        actual_path = find_doc(doc.system, doc.doc_type)
        
        if actual_path != expected_path:
            issues.append({
                'type': 'misplaced_doc',
                'doc': doc,
                'expected': expected_path,
                'actual': actual_path,
                'action': 'Move doc to correct location',
                'blocking': True  # BLOCKS chapter writing!
            })
    
    # Check 2: T0-T6 complete for referenced systems?
    for system in chapter_context.referenced_systems:
        for level in ['T0', 'T1', 'T2', 'T3', 'T4', 'T5', 'T6']:
            doc_path = f"knowledge_architecture/systems/{system}/{level}_*.md"
            if not exists(doc_path):
                issues.append({
                    'type': 'missing_t_level',
                    'system': system,
                    'level': level,
                    'action': f'Create {system}/{level} doc',
                    'blocking': (level in ['T0', 'T1', 'T2'])  # T0-T2 required!
                })
    
    # Check 3: System maps exist and current?
    for system in chapter_context.referenced_systems:
        map_path = f"knowledge_architecture/systems/{system}/system.map.lucid.json5"
        if not exists(map_path):
            issues.append({
                'type': 'missing_system_map',
                'system': system,
                'action': f'Create system map for {system}',
                'blocking': True
            })
        elif not is_current(map_path):
            issues.append({
                'type': 'outdated_system_map',
                'system': system,
                'action': f'Update system map for {system}',
                'blocking': False  # Warning, not blocking
            })
    
    # Check 4: SUPER_INDEX has all concepts?
    super_index = load_super_index()
    for concept in chapter_context.concepts:
        if concept not in super_index:
            issues.append({
                'type': 'missing_concept',
                'concept': concept,
                'action': f'Add {concept} to SUPER_INDEX',
                'blocking': False  # Can add during writing
            })
    
    # Check 5: Goals properly reflect systems?
    goal_tree = load_goal_tree()
    for system in chapter_context.referenced_systems:
        if not system_has_goal(system, goal_tree):
            issues.append({
                'type': 'system_without_goal',
                'system': system,
                'action': f'Add goal for {system} OR document why no goal',
                'blocking': False
            })
    
    # Categorize issues
    blocking_issues = [i for i in issues if i['blocking']]
    warning_issues = [i for i in issues if not i['blocking']]
    
    return ValidationResult(
        valid=(len(blocking_issues) == 0),
        blocking_issues=blocking_issues,
        warnings=warning_issues,
        must_fix_before_writing=blocking_issues,
        can_fix_during_writing=warning_issues
    )
```

**If blocking issues found: STOP, fix organization first, THEN write chapter!**
