## Usage

```typescript
import { IGODNField, simulate_igodn } from '@aimos/igodn';

// Initialize field
const field = new IGODNField();

// Add nodes
field.addNode({
  id: 'SAFETY',
  type: 'ANCHOR',
  cluster: 'SAFETY',
  position: { /* DualQuatPose */ },
  mass: 10.0,
  perimeter_radius: 1.0,
  // ...
});

// Run simulation
await simulate_igodn(field, {
  max_iterations: 1000,
  convergence_threshold: 1e-8
});
```
