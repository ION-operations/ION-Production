### Dependency Hashing

```python
def compute_dependency_hash(entry: IndexEntry) -> str:
    """Hash of all child content for change detection"""
    if not entry.child_ids:
        # Leaf node: hash own content
        return hashlib.sha256(entry.content_summary.encode()).hexdigest()
    
    # Non-leaf: hash all children
    child_entries = [get_entry(cid) for cid in entry.child_ids]
    child_hashes = sorted([c.dependency_hash for c in child_entries])
    
    canonical = json.dumps({
        "children": child_hashes,
        "atoms": sorted(entry.atom_refs)
    }, sort_keys=True)
    
    return hashlib.sha256(canonical.encode()).hexdigest()

def update_dependency_hashes():
    """Recompute hashes bottom-up when content changes"""
    # Start at bottom (level 6), propagate up
    for level_num in [6, 5, 4, 3, 2, 1]:
        level = levels[level_num]
        for entry in level.entries.values():
            entry.dependency_hash = compute_dependency_hash(entry)
```

### Complete Retrieval Pipeline

```python
def retrieve(
    query: str,
    config: RetrievalConfig
) -> RetrievalResult:
    """Two-stage retrieval with quality pipeline"""
    # Embed query
    query_embedding = embed(query)
    
    # === STAGE 1: COARSE RETRIEVAL ===
    candidates = coarse_retrieval(query, query_embedding, k=config.k_candidates)
    
    # === STAGE 2: REFINEMENT ===
    
    # Step 1: DVNS Physics Optimization
    if config.enable_dvns:
        optimized, iterations = physics_refinement(candidates, query_embedding)
        candidates = optimized
    else:
        iterations = 0
    
    # Step 2: Deduplication
    duplicates_removed = 0
    if config.enable_dedup:
        candidates, duplicates_removed = remove_duplicates(
            candidates,
            threshold=config.similarity_threshold
        )
    
    # Step 3: Conflict Resolution
    conflicts_resolved = 0
    if config.enable_conflict_resolution:
        conflicts = detect_conflicts(candidates)
        if conflicts:
            candidates, conflicts_resolved = resolve_conflicts(
                candidates,
                conflicts
            )
    
    # Step 4: Strategic Compression
    if config.enable_compression:
        candidates = apply_compression(candidates)
    
    # Step 5: Budget Fitting
    final_items = fit_to_budget(
        candidates,
        budget=config.token_budget,
        preserve_diversity=config.preserve_diversity
    )
    
    # Calculate RS-lift (improvement over baseline)
    rs_lift = calculate_rs_lift(final_items, baseline_items)
    
    return RetrievalResult(
        items=final_items,
        total_tokens=sum(item.tokens for item in final_items),
        items_count=len(final_items),
        rs_lift=rs_lift,
        dvns_applied=config.enable_dvns,
        iterations=iterations,
        duplicates_removed=duplicates_removed,
        conflicts_resolved=conflicts_resolved,
        metrics={...}
    )
```

## Deduplication Strategy

### Semantic Clustering

```python
def remove_duplicates(
    items: List[BudgetItem],
    threshold: float = 0.85
) -> Tuple[List[BudgetItem], int]:
    """Cluster and deduplicate semantically similar items"""
    if not items:
        return [], 0
    
    # Build similarity matrix
    embeddings = np.array([item.embedding for item in items])
    similarities = cosine_similarity(embeddings)
    
    # Cluster using threshold
    clusters = []
    assigned = set()
    
    for i, item in enumerate(items):
        if i in assigned:
            continue
        
        # Start new cluster
        cluster = [i]
        assigned.add(i)
        
        # Find similar items
        for j in range(i + 1, len(items)):
            if j not in assigned and similarities[i, j] >= threshold:
                cluster.append(j)
                assigned.add(j)
        
        clusters.append(cluster)
    
    # Select best from each cluster
    kept = []
    suppressed_count = 0
    
    for cluster_indices in clusters:
        cluster_items = [items[i] for i in cluster_indices]
        
        # Select best (highest relevance score)
        best = max(cluster_items, key=lambda x: x.relevance_score)
        kept.append(best)
        suppressed_count += len(cluster_items) - 1
    
    return kept, suppressed_count
```

```python
class BudgetError(Exception):
    """Raised when budget exceeded"""
    pass

def fit_to_budget_safe(
    items: List[BudgetItem],
    budget: int
) -> List[BudgetItem]:
    """Fit to budget with validation"""
    selected = fit_to_budget(items, budget)
    total = sum(item.tokens for item in selected)
    
    if total > budget:
        raise BudgetError(f"Budget exceeded: {total} > {budget}")
    
    return selected
```

```python
def compute_forces_parallel(particles: List[ContextParticle], query_embedding: List[float]) -> Dict[str, Vector]:
    """Compute forces in parallel for better performance."""
    forces = {}
```

### DVNS Force Computation

**Detailed Force Calculations:**

```python
def compute_gravity_force(
    particle: ContextParticle,
    query_embedding: List[float],
    other_particles: List[ContextParticle],
    config: DVNSConfig
) -> Vector:
    """
    Compute gravitational force attracting particle toward query.
    
    Formula:
    F_gravity = G * (m1 * m2 / r²) * direction_to_query
    
    Where:
    - G: Gravitational constant (config.gravity_strength)
    - m1, m2: Masses (relevance scores)
    - r: Distance to query in embedding space
    - direction_to_query: Unit vector toward query
    
    Returns:
        Vector: Gravitational force vector
    """
    # Compute distance to query
    query_vector = Vector.from_iterable(query_embedding[:3])  # Use first 3 dims
    direction = query_vector - particle.position
    distance = direction.magnitude()
    
    if distance < config.min_distance:
        distance = config.min_distance
    
    # Compute force magnitude
    mass = particle.mass
    query_mass = 1.0  # Query has unit mass
    force_magnitude = config.gravity_strength * (mass * query_mass) / (distance ** 2)
    
    # Apply force in direction toward query
    force = direction.normalize() * force_magnitude
    
    return force

def compute_elastic_force(
    particle: ContextParticle,
    parent_particle: Optional[ContextParticle],
    config: DVNSConfig
) -> Vector:
    """
    Compute elastic force maintaining hierarchical structure.
    
    Formula:
    F_elastic = -k * (r - r0) * direction_to_parent
    
    Where:
    - k: Spring constant (config.elastic_strength)
    - r: Current distance to parent
    - r0: Rest length (config.elastic_rest_length)
    
    Returns:
        Vector: Elastic force vector
    """
    if parent_particle is None:
        return Vector(0.0, 0.0, 0.0)
    
    # Compute distance to parent
    direction = particle.position - parent_particle.position
    distance = direction.magnitude()
    
    # Compute force magnitude
    displacement = distance - config.elastic_rest_length
    force_magnitude = -config.elastic_strength * displacement
    
    # Apply force
    force = direction.normalize() * force_magnitude
    
    return force

def compute_repulse_force(
    particle: ContextParticle,
    other: ContextParticle,
    config: DVNSConfig
) -> Vector:
    """
    Compute repulsive force separating similar particles.
    
    Formula:
    F_repulse = δ / (r² + ε) * direction_away
    
    Where:
    - δ: Repulse strength (config.repulse_strength)
    - r: Distance between particles
    - ε: Softening parameter (config.repulse_softening)
    
    Returns:
        Vector: Repulsive force vector
    """
    # Compute distance
    direction = particle.position - other.position
    distance = direction.magnitude()
    
    if distance < config.min_distance:
        distance = config.min_distance
    
    # Compute force magnitude
    force_magnitude = config.repulse_strength / (distance ** 2 + config.repulse_softening)
    
    # Apply force away from other particle
    force = direction.normalize() * force_magnitude
    
    return force

def compute_damping_force(
    particle: ContextParticle,
    config: DVNSConfig
) -> Vector:
    """
    Compute damping force stabilizing the system.
    
    Formula:
    F_damping = -c * v
    
    Where:
    - c: Damping coefficient (config.damping_coefficient)
    - v: Velocity vector
    
    Returns:
        Vector: Damping force vector
    """
    # Damping opposes velocity
    force = particle.velocity * (-config.damping_coefficient)
    
    return force
```

### Velocity-Verlet Integration

**Integration Algorithm:**

```python
def velocity_verlet_step(
    particles: List[ContextParticle],
    forces: Dict[str, Vector],
    config: DVNSConfig
) -> List[ContextParticle]:
    """
    Perform one Velocity-Verlet integration step.
    
    Algorithm:
    1. Update positions: x(t+dt) = x(t) + v(t)*dt + 0.5*a(t)*dt²
    2. Compute new accelerations: a(t+dt) = F(t+dt) / m
    3. Update velocities: v(t+dt) = v(t) + 0.5*(a(t) + a(t+dt))*dt
    
    Args:
        particles: Current particle states
        forces: Current force vectors
        config: DVNS configuration
    
    Returns:
        List[ContextParticle]: Updated particles
    """
    dt = config.time_step
    updated_particles = []
    
    # Store old accelerations
    old_accelerations = {p.id: p.acceleration for p in particles}
    
    # Step 1: Update positions
    for particle in particles:
        force = forces.get(particle.id, Vector(0.0, 0.0, 0.0))
        acceleration = force / particle.mass
        
        # Update position
        new_position = (
            particle.position +
            particle.velocity * dt +
            acceleration * (0.5 * dt * dt)
        )
        
        # Clamp position
        new_position = new_position.clamp(config.max_distance)
        
        particle.position = new_position
        particle.acceleration = acceleration
    
    # Step 2: Recompute forces at new positions
    new_forces = compute_forces(particles, query_embedding)
    
    # Step 3: Update velocities
    for particle in particles:
        new_force = new_forces.get(particle.id, Vector(0.0, 0.0, 0.0))
        new_acceleration = new_force / particle.mass
        
        old_accel = old_accelerations[particle.id]
        
        # Update velocity
        new_velocity = (
            particle.velocity +
            (old_accel + new_acceleration) * (0.5 * dt)
        )
        
        # Clamp velocity
        new_velocity = new_velocity.clamp(config.max_velocity_magnitude)
        
        particle.velocity = new_velocity
        particle.acceleration = new_acceleration
        
        updated_particles.append(particle)
    
    return updated_particles
```

### Deduplication Algorithm

**Semantic Clustering:**

```python
def semantic_clustering(
    items: List[BudgetItem],
    threshold: float = 0.85
) -> List[List[int]]:
    """
    Cluster items by semantic similarity.
    
    Algorithm:
    1. Build similarity matrix
    2. For each unassigned item:
       a. Create new cluster
       b. Find all items with similarity >= threshold
       c. Add to cluster
    3. Return cluster assignments
    
    Returns:
        List[List[int]]: Cluster indices
    """
    n = len(items)
    similarities = np.zeros((n, n))
    
    # Build similarity matrix
    for i in range(n):
        for j in range(i + 1, n):
            sim = cosine_similarity(
                items[i].embedding,
                items[j].embedding
            )
            similarities[i, j] = sim
            similarities[j, i] = sim
    
    # Cluster
    clusters = []
    assigned = set()
    
    for i in range(n):
        if i in assigned:
            continue
        
        # Start new cluster
        cluster = [i]
        assigned.add(i)
        
        # Find similar items
        for j in range(i + 1, n):
            if j not in assigned and similarities[i, j] >= threshold:
                cluster.append(j)
                assigned.add(j)
        
        clusters.append(cluster)
    
    return clusters

def select_cluster_representative(
    cluster_items: List[BudgetItem],
    recency_bias: float = 0.2,
    authority_bias: float = 0.2
) -> BudgetItem:
    """
    Select best representative from cluster.
    
    Scoring:
    score = (1 - recency_bias - authority_bias) * relevance +
            recency_bias * recency_score +
            authority_bias * authority_score
    
    Returns:
        BudgetItem: Best representative
    """
    relevance_weight = 1.0 - recency_bias - authority_bias
    
    scored_items = []
    for item in cluster_items:
        relevance_score = item.relevance_score * relevance_weight
        recency_score = item.metadata.get("recency_score", 0.5) * recency_bias
        authority_score = item.metadata.get("authority_score", 0.5) * authority_bias
        
        total_score = relevance_score + recency_score + authority_score
        
        scored_items.append((total_score, item))
    
    # Return best
    best = max(scored_items, key=lambda x: x[0])
    return best[1]
```

### Conflict Resolution Algorithm

**Contradiction Detection:**

```python
def detect_contradictions(
    items: List[BudgetItem]
) -> List[Tuple[str, List[BudgetItem], List[BudgetItem]]]:
    """
    Detect contradictory items (same topic, opposing stance).
    
    Returns:
        List[Tuple[topic, positive_items, negative_items]]: Contradiction clusters
    """
    # Group by topic
    topic_groups: Dict[str, Dict[str, List[BudgetItem]]] = {}
    
    for item in items:
        topic = extract_topic(item)
        stance = normalize_stance(item)
        
        if topic and stance:
            topic_groups.setdefault(topic, {}).setdefault(stance, []).append(item)
    
    # Find contradictions
    contradictions = []
    for topic, stance_map in topic_groups.items():
        if len(stance_map) > 1:
            # Multiple stances = contradiction
            positive_items = stance_map.get("positive", [])
            negative_items = stance_map.get("negative", [])
            
            if positive_items and negative_items:
                contradictions.append((topic, positive_items, negative_items))
    
    return contradictions

def resolve_contradiction(
    topic: str,
    positive_items: List[BudgetItem],
    negative_items: List[BudgetItem],
    recency_bias: float = 0.2,
    authority_bias: float = 0.3
) -> BudgetItem:
    """
    Resolve contradiction by selecting best stance.
    
    Selection criteria:
    1. Recency (newer information preferred)
    2. Authority (higher confidence preferred)
    3. Relevance (higher relevance preferred)
    
    Returns:
        BudgetItem: Best representative
    """
    # Score each stance
    positive_score = score_stance(positive_items, recency_bias, authority_bias)
    negative_score = score_stance(negative_items, recency_bias, authority_bias)
    
    # Select winning stance
    if positive_score > negative_score:
        winning_items = positive_items
        winning_stance = "positive"
    else:
        winning_items = negative_items
        winning_stance = "negative"
    
    # Select best item from winning stance
    best_item = select_cluster_representative(winning_items)
    
    return best_item
```

### Compression Algorithm

**Age-Based Compression:**

```python
def determine_compression_level(
    item: BudgetItem,
    config: CompressionConfig,
    reference_time: datetime
) -> CompressionLevel:
    """
    Determine compression level based on age and priority.
    
    Logic:
    - High priority items (priority > threshold): Always FULL
    - Age < full_detail_threshold: FULL
    - Age < detailed_summary_threshold: DETAILED
    - Age < brief_summary_threshold: BRIEF
    - Age >= brief_summary_threshold: REFERENCE
    
    Returns:
        CompressionLevel: Compression level to apply
    """
    # Check priority override
    priority = item.metadata.get("priority", 0.5)
    if priority > config.priority_threshold:
        return CompressionLevel.FULL
    
    # Determine age
    item_time = item.metadata.get("created_at", reference_time)
    age_days = (reference_time - item_time).days
    
    # Determine level
    if age_days < config.full_detail_threshold:
        return CompressionLevel.FULL
    elif age_days < config.detailed_summary_threshold:
        return CompressionLevel.DETAILED
    elif age_days < config.brief_summary_threshold:
        return CompressionLevel.BRIEF
    else:
        return CompressionLevel.REFERENCE

def compress_content(
    content: str,
    level: CompressionLevel,
    config: CompressionConfig
) -> str:
    """
    Compress content based on compression level.
    
    Strategies:
    - FULL: No compression
    - DETAILED: Extract key sentences (70% tokens)
    - BRIEF: Extract summary (40% tokens)
    - REFERENCE: Extract keywords only (15% tokens)
    
    Returns:
        str: Compressed content
    """
    if level == CompressionLevel.FULL:
        return content
    
    # Estimate tokens
    current_tokens = estimate_tokens(content)
    target_tokens = int(current_tokens * config.compression_ratio)
    
    if level == CompressionLevel.DETAILED:
        # Extract key sentences
        sentences = extract_sentences(content)
        key_sentences = extract_key_sentences(sentences, target_tokens)
        compressed = " ".join(key_sentences)
    
    elif level == CompressionLevel.BRIEF:
        # Generate summary
        compressed = generate_summary(content, target_tokens)
    
    elif level == CompressionLevel.REFERENCE:
        # Extract keywords
        keywords = extract_keywords(content)
        compressed = " ".join(keywords[:target_tokens // 2])
    
    # Ensure minimum tokens
    compressed_tokens = estimate_tokens(compressed)
    if compressed_tokens < config.min_compressed_tokens:
        compressed = content[:config.min_compressed_tokens * 4]  # Approximate
    
    return compressed
```

```python
        """Get shard for document."""
        shard_index = hash(doc_id) % self.shard_count
        return self.shards[shard_index]
    
    def index_document(self, content: str, doc_id: str, **kwargs) -> str:
        """Index document on appropriate shard."""
        shard = self.get_shard(doc_id)
        return shard.index_document(content, doc_id, **kwargs)
    
    def retrieve(self, query: str, **kwargs) -> RetrievalResult:
        """Retrieve from all shards and merge."""
        # Retrieve from all shards
        results = []
        for shard in self.shards:
            result = shard.retrieve(query, **kwargs)
            results.append(result)
        
        # Merge results
        merged_items = []
        for result in results:
            merged_items.extend(result.selected_items)
        
        # Deduplicate
        deduplicated, _ = deduplicate_candidates(merged_items, threshold=0.85)
        
        # Fit to budget
        allocation = budget_manager.optimize_for_budget(
            deduplicated,
            token_budget=kwargs.get("token_budget", 8000)
        )
        
        return RetrievalResult(
            selected_items=allocation.included,
            total_tokens=allocation.total_tokens_used,
            # ... other metrics
        )
```

### Caching Strategy

**Multi-Level Caching:**

```python
class MultiLevelCache:
    """Multi-level cache for HHNI operations."""
    
    def __init__(self):
        # L1: In-memory cache (fast, limited size)
        self.l1_cache: Dict[str, Any] = {}
        self.l1_max_size = 1000
        
        # L2: Redis cache (medium speed, larger size)
        self.l2_cache = RedisCache(ttl=3600)
        
        # L3: Database cache (slow, persistent)
        self.l3_cache = DatabaseCache()
    
    def get(self, key: str) -> Optional[Any]:
        """Get from cache (L1 -> L2 -> L3)."""
        # Try L1
        if key in self.l1_cache:
            return self.l1_cache[key]
        
        # Try L2
        value = self.l2_cache.get(key)
        if value:
            # Promote to L1
            self.l1_cache[key] = value
            return value
        
        # Try L3
        value = self.l3_cache.get(key)
        if value:
            # Promote to L2 and L1
            self.l2_cache.set(key, value)
            self.l1_cache[key] = value
            return value
        
        return None
    
    def set(self, key: str, value: Any):
        """Set in all cache levels."""
        self.l1_cache[key] = value
        self.l2_cache.set(key, value)
        self.l3_cache.set(key, value)
        
        # Evict L1 if over limit
        if len(self.l1_cache) > self.l1_max_size:
            # Remove oldest (LRU)
            oldest_key = next(iter(self.l1_cache))
            del self.l1_cache[oldest_key]
```
