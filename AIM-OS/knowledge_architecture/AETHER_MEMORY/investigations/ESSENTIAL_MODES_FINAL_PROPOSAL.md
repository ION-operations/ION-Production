### **After Grounding:**

```python
def select_next_mode_after_grounding():
    # Check last session state
    last_mode = get_last_session_mode()
    last_task = get_last_task()
    
    # If continuing work
    if last_task and last_task.incomplete:
        if last_task.type == "implementation":
            return "BUILDING"
        elif last_task.type == "investigation":
            return "THINKING"
        elif last_task.type == "audit":
            return "REVIEWING"
    
    # If starting new work
    if user_input_suggests_planning():
        return "PLANNING"
    elif user_input_suggests_discussion():
        return "COMMUNICATING"
    
    # Default: Communicating (safe start)
    return "COMMUNICATING"
```

```python
def detect_mode_change(user_input, current_mode):
    # Explicit mode requests
    if "/mode" in user_input:
        return parse_mode_request(user_input)
    
    # Keywords
    if "implement" or "build" or "create":
        return "BUILDING"

        return "THINKING"
    elif "plan" or "organize" or "prioritize":

        return "REVIEWING"  # Time to validate
```
